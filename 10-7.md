library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity pwm_fsm is
    generic (
        CLK_FREQ    : positive := 100000000;
        PWM_PERIOD  : positive := 100000;
        DUTY_STEPS  : positive := 10;
        HOLD_CYCLES : positive := 4
    );
    port (
        clk              : in  std_logic;
        rst_n            : in  std_logic;
        pwm_out          : out std_logic;
        on_counter_out   : out natural range 0 to PWM_PERIOD;
        off_counter_out  : out natural range 0 to PWM_PERIOD;
        hold_counter_out : out natural range 0 to HOLD_CYCLES - 1;
        duty_level_out   : out natural range 1 to DUTY_STEPS - 1
    );
end entity pwm_fsm;

architecture rtl of pwm_fsm is

    type state_type is (LED_ON, LED_OFF);

    constant HOLD_TICK : positive :=
        CLK_FREQ / (30 * HOLD_CYCLES);

    signal state : state_type := LED_OFF;

    signal on_counter   : natural range 0 to PWM_PERIOD := 0;
    signal off_counter  : natural range 0 to PWM_PERIOD := 0;
    signal time_counter : natural range 0 to HOLD_TICK - 1 := 0;

    signal hold_counter : natural range 0 to HOLD_CYCLES - 1 := 0;
    signal duty_level   : natural range 1 to DUTY_STEPS - 1 := 1;

    signal direction : std_logic := '1';

    signal on_time  : natural range 0 to PWM_PERIOD;
    signal off_time : natural range 0 to PWM_PERIOD;

begin

    on_time <= (PWM_PERIOD * duty_level) / DUTY_STEPS;
    off_time <= PWM_PERIOD - on_time;

    process(clk, rst_n)
    begin
        if rst_n = '0' then
            state <= LED_OFF;

        elsif rising_edge(clk) then

            case state is

                when LED_ON =>

                    if on_counter >= on_time - 1 then
                        state <= LED_OFF;
                    end if;

                when LED_OFF =>

                    if off_counter >= off_time - 1 then
                        state <= LED_ON;
                    end if;

            end case;

        end if;
    end process;


    process(clk, rst_n)
    begin
        if rst_n = '0' then
            on_counter <= 0;

        elsif rising_edge(clk) then

            if state = LED_ON then

                if on_counter >= on_time - 1 then
                    on_counter <= 0;
                else
                    on_counter <= on_counter + 1;
                end if;

            else
                on_counter <= 0;
            end if;

        end if;
    end process;


    process(clk, rst_n)
    begin
        if rst_n = '0' then
            off_counter <= 0;

        elsif rising_edge(clk) then

            if state = LED_OFF then

                if off_counter >= off_time - 1 then
                    off_counter <= 0;
                else
                    off_counter <= off_counter + 1;
                end if;

            else
                off_counter <= 0;
            end if;

        end if;
    end process;


    process(clk, rst_n)
    begin
        if rst_n = '0' then
            time_counter <= 0;

        elsif rising_edge(clk) then

            if time_counter >= HOLD_TICK - 1 then
                time_counter <= 0;
            else
                time_counter <= time_counter + 1;
            end if;

        end if;
    end process;


    process(clk, rst_n)
    begin
        if rst_n = '0' then
            hold_counter <= 0;

        elsif rising_edge(clk) then

            if time_counter >= HOLD_TICK - 1 then

                if hold_counter >= HOLD_CYCLES - 1 then
                    hold_counter <= 0;
                else
                    hold_counter <= hold_counter + 1;
                end if;

            end if;

        end if;
    end process;


    process(clk, rst_n)
    begin
        if rst_n = '0' then
            duty_level <= 1;

        elsif rising_edge(clk) then

            if time_counter >= HOLD_TICK - 1 and
               hold_counter >= HOLD_CYCLES - 1 then

                if direction = '1' then

                    if duty_level < DUTY_STEPS - 1 then
                        duty_level <= duty_level + 1;
                    end if;

                else

                    if duty_level > 1 then
                        duty_level <= duty_level - 1;
                    end if;

                end if;

            end if;

        end if;
    end process;


    process(clk, rst_n)
    begin
        if rst_n = '0' then
            direction <= '1';

        elsif rising_edge(clk) then

            if time_counter >= HOLD_TICK - 1 and
               hold_counter >= HOLD_CYCLES - 1 then

                if direction = '1' and
                   duty_level >= DUTY_STEPS - 1 then

                    direction <= '0';

                elsif direction = '0' and
                      duty_level <= 1 then

                    direction <= '1';

                end if;

            end if;

        end if;
    end process;


    pwm_out <= '1' when state = LED_ON else '0';

    on_counter_out <= on_counter;
    off_counter_out <= off_counter;
    hold_counter_out <= hold_counter;
    duty_level_out <= duty_level;

end architecture rtl;
