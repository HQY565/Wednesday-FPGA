**時脈除頻器 (clk_div.vhd)**
``` vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_UNSIGNED.ALL;

entity clk_div is
    Port (
        clk_100m : in STD_LOGIC;
        rst      : in STD_LOGIC;
        clk_25m  : out STD_LOGIC
    );
end clk_div;

architecture Behavioral of clk_div is
    signal count : STD_LOGIC_VECTOR(1 downto 0) := (others => '0');
    signal temp_clk : STD_LOGIC := '0';
begin
    process(clk_100m, rst)
    begin
        if rst = '1' then
            count <= (others => '0');
            temp_clk <= '0';
        elsif rising_edge(clk_100m) then
            if count = "01" then -- 100MHz 轉 25MHz (每 2 個週期翻轉一次)
                temp_clk <= not temp_clk;
                count <= (others => '0');
            else
                count <= count + 1;
            end if;
        end if;
    end process;
    
    clk_25m <= temp_clk;
end Behavioral;
```

**VGA 驅動模組 (vga_driver.vhd)**
``` vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.all;
use IEEE.STD_LOGIC_ARITH.all;
use IEEE.STD_LOGIC_UNSIGNED.all;

entity vga_driver is
    port(
        clock : in std_logic; -- 25 MHz 像素時脈
        red_in, green_in, blue_in : in std_logic;
        row, column : out std_logic_vector(9 downto 0);
        Rout, Gout, Bout, H, V : out std_logic
    );
end vga_driver;

architecture behavioral of vga_driver is
    subtype counter is std_logic_vector(9 downto 0);
    constant B : natural := 93; -- 水平同步脈衝 (Horizontal sync)
    constant C : natural := 45; -- 前廊 (Front porch)
    constant D : natural := 640; -- 顯示區像素 (Active columns)
    constant E : natural := 22; -- 後廊 (Back porch)
    constant A : natural := B + C + D + E; -- 總計 800
    
    constant P : natural := 2; -- 垂直同步脈衝 (Vertical sync)
    constant Q : natural := 32; -- 前廊
    constant R : natural := 480; -- 顯示區列數 (Active rows)
    constant S : natural := 11; -- 後廊
    constant O : natural := P + Q + R + S; -- 總計 525
begin
    Rout <= red_in;
    Gout <= green_in;
    Bout <= blue_in;
    
    process
        variable vertical, horizontal : counter := (others => '0');
    begin
        wait until rising_edge(clock);
        
        if horizontal < A - 1 then
            horizontal := horizontal + 1;
        else
            horizontal := (others => '0');
            if vertical < O - 1 then
                vertical := vertical + 1;
            else
                vertical := (others => '0');
            end if;
        end if;

        -- 產生 HSYNC (負脈衝)
        if horizontal >= (D + E) and horizontal < (D + E + B) then
            H <= '0';
        else
            H <= '1';
        end if;

        -- 產生 VSYNC (負脈衝)
        if vertical >= (R + S) and vertical < (R + S + P) then
            V <= '0';
        else
            V <= '1';
        end if;

        row <= vertical;
        column <= horizontal;
    end process;
end architecture;
```
**頂層模組與圖形邏輯 (vga_top.vhd)**
``` vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_ARITH.ALL;
use IEEE.STD_LOGIC_UNSIGNED.ALL;

entity vga_top is
    Port (
        clk : in STD_LOGIC; -- 100MHz 系統時脈
        rst : in STD_LOGIC;
        H, V : out STD_LOGIC; -- 接至螢幕的 HSYNC, VSYNC
        R, G, B : out STD_LOGIC -- 接至螢幕的 RGB 信號
    );
end vga_top;

architecture Behavioral of vga_top is
    signal pixel_clk : STD_LOGIC;
    signal row, column : STD_LOGIC_VECTOR(9 downto 0);
    signal red_signal, green_signal, blue_signal : STD_LOGIC;
    
    -- 1. 圓形參數：圓心 (160, 240)，半徑 60 (r^2 = 3600)
    constant CIRCLE_X : integer := 160;
    constant CIRCLE_Y : integer := 240;
    constant R_SQUARE : integer := 3600;

    -- 2. 正方形參數：左上角 (280, 180)，邊長 120
    constant SQUARE_X1 : integer := 280;
    constant SQUARE_X2 : integer := 400; -- 280 + 120
    constant SQUARE_Y1 : integer := 180;
    constant SQUARE_Y2 : integer := 300; -- 180 + 120

    -- 3. 三角形參數（以三個頂點或邊界直線判斷）：
    -- 這裡以簡單的等腰三角形為例：頂點 (540, 150)，底邊由 y=300 到 y=300 且 x 從 480 到 600
    -- 簡化成邊界條件：y 在 150 到 300 之間，且左右邊界隨高度遞增/遞減
begin
    -- 1. 實例化時脈除頻器 (100MHz ➔ 25MHz)
    clk_div_inst : entity work.clk_div
        port map (
            clk_100m => clk,
            rst      => rst,
            clk_25m  => pixel_clk
        );

    -- 2. 實例化 VGA 驅動器
    vga_driver_inst : entity work.vga_driver
        port map (
            clock    => pixel_clk,
            red_in   => red_signal,
            green_in => green_signal,
            blue_in  => blue_signal,
            row      => row,
            column   => column,
            Rout     => R,
            Gout     => G,
            Bout     => B,
            H        => H,
            V        => V
        );

    -- 3. 多圖形繪製邏輯 (Icon Logic)
    process(row, column)
        variable x_pos, y_pos : integer;
        variable c_x_diff, c_y_diff : integer;
        variable in_circle, in_square, in_triangle : boolean;
    begin
        x_pos := conv_integer(column);
        y_pos := conv_integer(row);
        
        -- 初始化布林變數
        in_circle := false;
        in_square := false;
        in_triangle := false;

        -- 確保在 640x480 有效顯示區內
        if x_pos < 640 and y_pos < 480 then
            
            -- [圖形 1：圓形] 判斷公式 (x - h)^2 + (y - k)^2 <= r^2
            c_x_diff := x_pos - CIRCLE_X;
            c_y_diff := y_pos - CIRCLE_Y;
            if (c_x_diff * c_x_diff + c_y_diff * c_y_diff) <= R_SQUARE then
                in_circle := true;
            end if;

            -- [圖形 2：正方形] 判斷 X 與 Y 的上下左右邊界
            if x_pos >= SQUARE_X1 and x_pos <= SQUARE_X2 and 
               y_pos >= SQUARE_Y1 and y_pos <= SQUARE_Y2 then
                in_square := true;
            end if;

            -- [圖形 3：三角形] 依據白板上的直線方程式原理（例如邊界斜率限制）
            -- 這裡設定頂點 (540, 150)，底邊 y 從 180 到 300，寬度從 480 到 600
            if y_pos >= 180 and y_pos <= 300 then
                -- 左斜邊與右斜邊的範圍限制
                if x_pos >= (540 - (y_pos - 180)) and x_pos <= (540 + (y_pos - 180)) then
                    in_triangle := true;
                end if;
            end if;

            -- 顏色輸出分配（優先順序：圓形 ➔ 紅色，正方形 ➔ 綠色，三角形 ➔ 藍色，其餘 ➔ 背景黑）
            if in_circle then
                red_signal   <= '1';
                green_signal <= '0';
                blue_signal  <= '0'; -- 紅色圓形
            elsif in_square then
                red_signal   <= '0';
                green_signal <= '1';
                blue_signal  <= '0'; -- 綠色正方形
            elsif in_triangle then
                red_signal   <= '0';
                green_signal <= '0';
                blue_signal  <= '1'; -- 藍色三角形
            else
                red_signal   <= '0';
                green_signal <= '0';
                blue_signal  <= '0'; -- 黑色背景
            end if;

        else
            -- 空白區間關閉輸出
            red_signal   <= '0';
            green_signal <= '0';
            blue_signal  <= '0';
        end if;
    end process;

end Behavioral;
```

``` vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity vga_top_tb is
end vga_top_tb;

architecture Behavioral of vga_top_tb is

    -- 宣告待測元件 (UUT: Unit Under Test)
    component vga_top
        Port (
            clk : in STD_LOGIC;
            rst : in STD_LOGIC;
            H   : out STD_LOGIC;
            V   : out STD_LOGIC;
            R   : out STD_LOGIC;
            G   : out STD_LOGIC;
            B   : out STD_LOGIC
        );
    end component;

    -- 測試信號宣告
    signal clk : STD_LOGIC := '0';
    signal rst : STD_LOGIC := '1';
    signal H   : STD_LOGIC;
    signal V   : STD_LOGIC;
    signal R   : STD_LOGIC;
    signal G   : STD_LOGIC;
    signal B   : STD_LOGIC;

    -- 定義 100MHz 時脈週期 (10 ns)
    constant CLK_PERIOD : time := 10 ns;

begin

    -- 實例化 Top 模組 (已修正符號)
    uut: vga_top
        port map (
            clk => clk,
            rst => rst,
            H   => H,
            V   => V,
            R   => R,
            G   => G,
            B   => B
        );

    -- 時脈產生製程 (Clock Generation)
    clk_process : process
    begin
        clk <= '0';
        wait for CLK_PERIOD / 2;
        clk <= '1';
        wait for CLK_PERIOD / 2;
    end process;

    -- 測試激發序列 (Stimulus Process)
    stim_proc: process
    begin
        -- 給予短暫重置信號
        rst <= '1';
        wait for 50 ns;
        rst <= '0';
        
        -- 讓模擬執行一段時間
        wait for 2 ms;
        
        -- 結束模擬
        wait;
    end process;

end Behavioral;
```
