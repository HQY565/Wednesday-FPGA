**乒乓球程式碼(pingpong.vhd)**
``` vhdl

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity pingpong is
    generic (
        CLK_FREQ   : positive := 100000000;
        MOVE_HZ    : positive := 4;
        RESULT_SEC : positive := 2
    );
    port (
        clk     : in  std_logic;
        rst     : in  std_logic;
        sw_l    : in  std_logic;
        sw_r    : in  std_logic;

        led     : out std_logic_vector(7 downto 0);
        score_l : out std_logic_vector(3 downto 0);
        score_r : out std_logic_vector(3 downto 0)
    );
end pingpong;

architecture Behavioral of pingpong is

    type state_type is (
        RIGHT_MOVING,
        LEFT_MOVING,
        LEFT_WIN,
        RIGHT_WIN
    );

    signal state : state_type := RIGHT_MOVING;

    constant MOVE_COUNT : positive := CLK_FREQ / MOVE_HZ;
    constant RESULT_COUNT : positive := CLK_FREQ * RESULT_SEC;

    signal move_counter : integer range 0 to MOVE_COUNT-1 := 0;
    signal result_counter : integer range 0 to RESULT_COUNT-1 := 0;

    signal ball_pos : integer range 0 to 7 := 0;

    signal left_score  : unsigned(3 downto 0) := (others => '0');
    signal right_score : unsigned(3 downto 0) := (others => '0');

    signal left_meta  : std_logic := '0';
    signal left_sync  : std_logic := '0';
    signal right_meta : std_logic := '0';
    signal right_sync : std_logic := '0';

    signal left_prev  : std_logic := '0';
    signal right_prev : std_logic := '0';

    signal left_press  : std_logic;
    signal right_press : std_logic;

begin

    -- Two-stage input synchronizers
    process(clk)
    begin
        if rising_edge(clk) then
            left_meta  <= sw_l;
            left_sync  <= left_meta;
            right_meta <= sw_r;
            right_sync <= right_meta;

            left_prev  <= left_sync;
            right_prev <= right_sync;
        end if;
    end process;

    -- Detect rising edge of buttons
    left_press  <= left_sync and not left_prev;
    right_press <= right_sync and not right_prev;

    -- Main FSM
    process(clk)
    begin
        if rising_edge(clk) then

            if rst = '1' then
                state <= RIGHT_MOVING;

                ball_pos <= 0;
                move_counter <= 0;
                result_counter <= 0;

                left_score  <= (others => '0');
                right_score <= (others => '0');

            else

                case state is

                    -- Ball moves from left to right
                    when RIGHT_MOVING =>

                        result_counter <= 0;

                        -- Right player presses too early
                        if right_press = '1' and ball_pos /= 7 then
                            state <= LEFT_WIN;
                            left_score <= left_score + 1;
                            move_counter <= 0;

                        -- Right player hits at position 7
                        elsif right_press = '1' and ball_pos = 7 then
                            state <= LEFT_MOVING;
                            move_counter <= 0;

                        elsif move_counter = MOVE_COUNT-1 then
                            move_counter <= 0;

                            -- Right player misses the ball
                            if ball_pos = 7 then
                                state <= LEFT_WIN;
                                left_score <= left_score + 1;
                            else
                                ball_pos <= ball_pos + 1;
                            end if;

                        else
                            move_counter <= move_counter + 1;
                        end if;

                    -- Ball moves from right to left
                    when LEFT_MOVING =>

                        result_counter <= 0;

                        -- Left player presses too early
                        if left_press = '1' and ball_pos /= 0 then
                            state <= RIGHT_WIN;
                            right_score <= right_score + 1;
                            move_counter <= 0;

                        -- Left player hits at position 0
                        elsif left_press = '1' and ball_pos = 0 then
                            state <= RIGHT_MOVING;
                            move_counter <= 0;

                        elsif move_counter = MOVE_COUNT-1 then
                            move_counter <= 0;

                            -- Left player misses the ball
                            if ball_pos = 0 then
                                state <= RIGHT_WIN;
                                right_score <= right_score + 1;
                            else
                                ball_pos <= ball_pos - 1;
                            end if;

                        else
                            move_counter <= move_counter + 1;
                        end if;

                    -- Left player wins this round
                    when LEFT_WIN =>

                        move_counter <= 0;

                        if result_counter = RESULT_COUNT-1 then
                            result_counter <= 0;
                            ball_pos <= 0;
                            state <= RIGHT_MOVING;
                        else
                            result_counter <= result_counter + 1;
                        end if;

                    -- Right player wins this round
                    when RIGHT_WIN =>

                        move_counter <= 0;

                        if result_counter = RESULT_COUNT-1 then
                            result_counter <= 0;
                            ball_pos <= 7;
                            state <= LEFT_MOVING;
                        else
                            result_counter <= result_counter + 1;
                        end if;

                end case;

            end if;

        end if;
    end process;

    -- LED output
    process(state, ball_pos)
        variable temp_led : std_logic_vector(7 downto 0);
    begin
        temp_led := (others => '0');

        case state is

            when RIGHT_MOVING | LEFT_MOVING =>
                temp_led(ball_pos) := '1';

            when LEFT_WIN | RIGHT_WIN =>
                temp_led := (others => '1');

        end case;

        led <= temp_led;
    end process;

    -- Score output
    score_l <= std_logic_vector(left_score);
    score_r <= std_logic_vector(right_score);

end Behavioral;
```

``` vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity pingpong_tb is
end pingpong_tb;

architecture Behavioral of pingpong_tb is

    -- Simulation clock: 100 MHz
    constant CLK_PERIOD : time := 10 ns;

    -- Speed up simulation
    constant TEST_CLK_FREQ : positive := 100;
    constant TEST_MOVE_HZ  : positive := 10;

    signal clk     : std_logic := '0';
    signal rst     : std_logic := '0';
    signal sw_l    : std_logic := '0';
    signal sw_r    : std_logic := '0';

    signal led     : std_logic_vector(7 downto 0);
    signal score_l : std_logic_vector(3 downto 0);
    signal score_r : std_logic_vector(3 downto 0);

begin

    -- Clock generation
    clk <= not clk after CLK_PERIOD / 2;

    -- DUT: Device Under Test
    uut : entity work.pingpong
        generic map (
            CLK_FREQ   => TEST_CLK_FREQ,
            MOVE_HZ    => TEST_MOVE_HZ,
            RESULT_SEC => 1
        )
        port map (
            clk     => clk,
            rst     => rst,
            sw_l    => sw_l,
            sw_r    => sw_r,
            led     => led,
            score_l => score_l,
            score_r => score_r
        );

    -- Test process
    stim_proc : process
    begin

        -- Reset
        rst <= '1';
        wait for 50 ns;
        rst <= '0';
        wait for 20 ns;

        report "TEST 1: Ball moving right";

        -- Ball moves from LED0 to LED7
        wait for 700 ns;

        report "TEST 2: Right player hits the ball";

        -- Right player presses button
        sw_r <= '1';
        wait for 40 ns;
        sw_r <= '0';

        -- Ball moves toward left
        wait for 700 ns;

        report "TEST 3: Left player hits the ball";

        -- Left player presses button
        sw_l <= '1';
        wait for 40 ns;
        sw_l <= '0';

        wait for 200 ns;

        report "TEST 4: Right player presses too early";

        sw_r <= '1';
        wait for 40 ns;
        sw_r <= '0';

        -- Wait for round reset
        wait for 1200 ns;

        report "TEST 5: Right player misses the ball";

        -- Do not press right button
        wait for 1000 ns;

        report "TEST 6: Check scores";

        report "Left Score = " &
            integer'image(to_integer(unsigned(score_l)));

        report "Right Score = " &
            integer'image(to_integer(unsigned(score_r)));

        report "Simulation finished.";

        wait;

    end process;

end Behavioral;
```
