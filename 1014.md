### pwm
``` vhdl
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity pwm_fsm is
    generic (
        CLK_FREQ    : positive := 100000000;
        DUTY_STEPS  : positive := 100;
        HOLD_CYCLES : positive := 4
    );
    port (
        clk              : in  std_logic;
        rst              : in  std_logic;
        pwm_out          : out std_logic;
        on_counter_out   : out natural;
        off_counter_out  : out natural;
        hold_counter_out : out natural range 0 to HOLD_CYCLES - 1;
        duty_level_out   : out natural range 0 to DUTY_STEPS
    );
end entity pwm_fsm;

architecture rtl of pwm_fsm is

    type state_type is (LED_ON, LED_OFF);

    constant PWM_PERIOD : positive :=
        CLK_FREQ / (30 * HOLD_CYCLES);

    signal state : state_type := LED_OFF;

    signal on_counter  : natural range 0 to PWM_PERIOD := 0;
    signal off_counter : natural range 0 to PWM_PERIOD := 0;

    signal hold_counter : natural range 0 to HOLD_CYCLES - 1 := 0;
    signal duty_level   : natural range 0 to DUTY_STEPS := 0;

    signal direction : std_logic := '1';

    signal on_time  : natural range 0 to PWM_PERIOD;
    signal off_time : natural range 0 to PWM_PERIOD;

    signal pwm_done : std_logic;

    signal next_duty_level : natural range 0 to DUTY_STEPS;

begin

    assert DUTY_STEPS >= 2 and PWM_PERIOD >= DUTY_STEPS
        report "Invalid generic values"
        severity failure;

    on_time <= (PWM_PERIOD / DUTY_STEPS) * duty_level
               + ((PWM_PERIOD mod DUTY_STEPS) * duty_level) / DUTY_STEPS;

    off_time <= PWM_PERIOD - on_time;

    process(duty_level, direction)
    begin
        next_duty_level <= duty_level;

        if direction = '1' then
            if duty_level < DUTY_STEPS then
                next_duty_level <= duty_level + 1;
            else
                next_duty_level <= duty_level - 1;
            end if;
        else
            if duty_level > 0 then
                next_duty_level <= duty_level - 1;
            else
                next_duty_level <= duty_level + 1;
            end if;
        end if;
    end process;

    pwm_done <= '1' when
        (state = LED_OFF and
         off_time > 0 and
         off_counter = off_time - 1) or
        (state = LED_ON and
         off_time = 0 and
         on_counter = on_time - 1)
        else '0';

    process(clk, rst)
    begin
        if rst = '1' then
            state <= LED_OFF;

        elsif rising_edge(clk) then

            case state is

                when LED_ON =>

                    if on_time = 0 then
                        state <= LED_OFF;

                    elsif on_counter >= on_time - 1 then

                        if off_time = 0 then

                            if pwm_done = '1' and
                               hold_counter = HOLD_CYCLES - 1 then

                                if next_duty_level = 0 then
                                    state <= LED_OFF;
                                else
                                    state <= LED_ON;
                                end if;

                            else
                                state <= LED_ON;
                            end if;

                        else
                            state <= LED_OFF;
                        end if;

                    end if;

                when LED_OFF =>

                    if off_time = 0 then
                        state <= LED_ON;

                    elsif off_counter >= off_time - 1 then

                        if pwm_done = '1' and
                           hold_counter = HOLD_CYCLES - 1 then

                            if next_duty_level = 0 then
                                state <= LED_OFF;
                            else
                                state <= LED_ON;
                            end if;

                        else

                            if on_time = 0 then
                                state <= LED_OFF;
                            else
                                state <= LED_ON;
                            end if;

                        end if;

                    end if;

            end case;

        end if;
    end process;

    process(clk, rst)
    begin
        if rst = '1' then
            on_counter <= 0;

        elsif rising_edge(clk) then

            if state = LED_ON and on_time > 0 then

                if on_counter >= on_time - 1 then
                    on_counter <= 0;
                else
                    on_counter <= on_counter + 1;
                end if;

            else
                on_counter <= 0;
            end if;

        end if;
    end process;

    process(clk, rst)
    begin
        if rst = '1' then
            off_counter <= 0;

        elsif rising_edge(clk) then

            if state = LED_OFF and off_time > 0 then

                if off_counter >= off_time - 1 then
                    off_counter <= 0;
                else
                    off_counter <= off_counter + 1;
                end if;

            else
                off_counter <= 0;
            end if;

        end if;
    end process;

    process(clk, rst)
    begin
        if rst = '1' then
            hold_counter <= 0;

        elsif rising_edge(clk) then

            if pwm_done = '1' then

                if hold_counter = HOLD_CYCLES - 1 then
                    hold_counter <= 0;
                else
                    hold_counter <= hold_counter + 1;
                end if;

            end if;

        end if;
    end process;

    process(clk, rst)
    begin
        if rst = '1' then
            duty_level <= 0;

        elsif rising_edge(clk) then

            if pwm_done = '1' and
               hold_counter = HOLD_CYCLES - 1 then

                duty_level <= next_duty_level;

            end if;

        end if;
    end process;

    process(clk, rst)
    begin
        if rst = '1' then
            direction <= '1';

        elsif rising_edge(clk) then

            if pwm_done = '1' and
               hold_counter = HOLD_CYCLES - 1 then

                if direction = '1' and
                   duty_level = DUTY_STEPS then
                    direction <= '0';

                elsif direction = '0' and
                      duty_level = 0 then
                    direction <= '1';
                end if;

            end if;

        end if;
    end process;

    pwm_out <= '1' when
        rst = '0' and state = LED_ON and on_time > 0
        else '0';

    on_counter_out   <= on_counter;
    off_counter_out  <= off_counter;
    hold_counter_out <= hold_counter;
    duty_level_out   <= duty_level;

end architecture rtl;
```

## tb

``` vhdl
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity pwm_fsm_tb is
end entity pwm_fsm_tb;

architecture sim of pwm_fsm_tb is

    constant CLK_PERIOD : time := 10 ns;

    signal clk     : std_logic := '0';
    signal rst     : std_logic := '1';
    signal pwm_out : std_logic;

    signal on_counter_out   : natural;
    signal off_counter_out  : natural;
    signal hold_counter_out : natural range 0 to 3;
    signal duty_level_out   : natural range 0 to 100;

begin

    uut : entity work.pwm_fsm
        generic map (
            CLK_FREQ    => 12000,
            DUTY_STEPS  => 100,
            HOLD_CYCLES => 4
        )
        port map (
            clk              => clk,
            rst              => rst,
            pwm_out          => pwm_out,
            on_counter_out   => on_counter_out,
            off_counter_out  => off_counter_out,
            hold_counter_out => hold_counter_out,
            duty_level_out   => duty_level_out
        );

    process
    begin
        clk <= '0';
        wait for CLK_PERIOD / 2;
        clk <= '1';
        wait for CLK_PERIOD / 2;
    end process;

    process
    begin
        rst <= '1';
        wait for 20 ns;
        rst <= '0';
        wait;
    end process;

    process
        variable previous_level : natural := 0;
        variable reached_100    : boolean := false;
        variable cycle_count    : natural := 0;
    begin
        wait until rst = '0';

        loop
            wait until rising_edge(clk);
            wait for 1 ns;

            if duty_level_out = 100 then
                reached_100 := true;
            end if;

            if reached_100 and
               previous_level = 1 and
               duty_level_out = 0 then

                cycle_count := cycle_count + 1;

                report "Completed PWM brightness cycle: " &
                       integer'image(cycle_count);

                reached_100 := false;

            end if;

            previous_level := duty_level_out;
        end loop;
    end process;

end architecture sim;
```
